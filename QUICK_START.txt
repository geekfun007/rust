🎯 核心功能：
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  在 Python 中使用：

    import rust_py_example as rpe

    # 用户管理
    user = rpe.PyUser("Alice", 25, "alice@example.com")
    print(user.description())
    print(user.is_adult())

    # 数学计算
    print(rpe.fibonacci(10))    # 55
    print(rpe.is_prime(17))     # True
    print(rpe.factorial(5))     # 120

    # 字符串处理
    print(rpe.reverse_string("hello"))
    print(rpe.word_count("hello world"))

    # 批量处理
    print(rpe.fibonacci_batch([1, 2, 3, 4, 5]))
